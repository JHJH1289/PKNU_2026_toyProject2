-- Travel Share PostgreSQL setup
-- Run as a PostgreSQL superuser:
--   psql -U postgres -f psql.txt
--
-- This project now uses a new database so the existing drive_db used by
-- another project is not deleted or modified.
--
-- Current app config expects:
--   database: travel_share_db
--   username: postgres
--   password: 12341234

-- 1. Create the new database if it does not exist.
SELECT 'CREATE DATABASE travel_share_db'
WHERE NOT EXISTS (
  SELECT FROM pg_database WHERE datname = 'travel_share_db'
)\gexec

-- 2. Connect to the new application database.
\connect travel_share_db

-- 3. Create User table.
CREATE TABLE IF NOT EXISTS users (
  id BIGSERIAL PRIMARY KEY,
  username VARCHAR(50) NOT NULL,
  password VARCHAR(255) NOT NULL,
  role VARCHAR(20),
  CONSTRAINT uk_users_username UNIQUE (username)
);

-- 4. Create Admin table.
CREATE TABLE IF NOT EXISTS admins (
  id BIGSERIAL PRIMARY KEY,
  username VARCHAR(50) NOT NULL,
  created_at TIMESTAMP NOT NULL,
  CONSTRAINT uk_admins_username UNIQUE (username)
);

-- 5. Create Post table.
CREATE TABLE IF NOT EXISTS posts (
  id BIGSERIAL PRIMARY KEY,
  owner_id VARCHAR(50) NOT NULL,
  image_storage_key VARCHAR(500) NOT NULL,
  content_type VARCHAR(255) NOT NULL,
  file_size BIGINT NOT NULL,
  caption VARCHAR(2000),
  location_name VARCHAR(120),
  created_at TIMESTAMP NOT NULL,
  view_count BIGINT NOT NULL DEFAULT 0
);

-- 6. Create comments table.
CREATE TABLE IF NOT EXISTS comments (
  id BIGSERIAL PRIMARY KEY,
  post_id BIGINT NOT NULL REFERENCES posts(id) ON DELETE CASCADE,
  owner_id VARCHAR(50) NOT NULL,
  content VARCHAR(1000) NOT NULL,
  created_at TIMESTAMP NOT NULL
);

-- 7. Create likes table.
CREATE TABLE IF NOT EXISTS post_likes (
  id BIGSERIAL PRIMARY KEY,
  post_id BIGINT NOT NULL REFERENCES posts(id) ON DELETE CASCADE,
  owner_id VARCHAR(50) NOT NULL,
  created_at TIMESTAMP NOT NULL,
  CONSTRAINT uk_post_likes_post_owner UNIQUE (post_id, owner_id)
);

-- 8. Create views table. One signed-in user counts only once per post.
CREATE TABLE IF NOT EXISTS post_views (
  id BIGSERIAL PRIMARY KEY,
  post_id BIGINT NOT NULL REFERENCES posts(id) ON DELETE CASCADE,
  viewer_id VARCHAR(50) NOT NULL,
  created_at TIMESTAMP NOT NULL,
  CONSTRAINT uk_post_views_post_viewer UNIQUE (post_id, viewer_id)
);

-- 9. Helpful indexes for feed, my page, comments, likes, views, and admin mode.
CREATE INDEX IF NOT EXISTS idx_posts_created_at
  ON posts (created_at DESC, id DESC);

CREATE INDEX IF NOT EXISTS idx_posts_owner_created_at
  ON posts (owner_id, created_at DESC, id DESC);

CREATE INDEX IF NOT EXISTS idx_comments_post_created_at
  ON comments (post_id, created_at, id);

CREATE INDEX IF NOT EXISTS idx_post_likes_post
  ON post_likes (post_id);

CREATE INDEX IF NOT EXISTS idx_post_likes_owner
  ON post_likes (owner_id);

CREATE INDEX IF NOT EXISTS idx_post_views_post
  ON post_views (post_id);

CREATE INDEX IF NOT EXISTS idx_post_views_viewer
  ON post_views (viewer_id);

-- 10. Verify created tables.
\dt
